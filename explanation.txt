TradeSimulator — System Design Review
=======================================

1. PostgreSQL Schema
--------------------

-- Extensions
CREATE EXTENSION IF NOT EXISTS "pgcrypto";

-- 1. USERS
-- Reasoning: separate auth from financial data; UUID prevents enumeration
CREATE TABLE users (
    id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    email       TEXT NOT NULL UNIQUE,
    password_hash TEXT NOT NULL,
    name        TEXT NOT NULL,
    created_at  TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at  TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 2. PORTFOLIOS (1:1 with user)
-- Reasoning: a user owns exactly one portfolio; keeps portfolio metadata separate
-- from user auth row to reduce lock contention during frequent P&L updates
CREATE TABLE portfolios (
    id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id     UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    -- cooldown_expires_at tracks when reset is next allowed
    cooldown_expires_at TIMESTAMPTZ,
    created_at  TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
CREATE UNIQUE INDEX idx_portfolios_user ON portfolios(user_id);

-- 3. PORTFOLIO BALANCES (one row per currency per portfolio)
-- Reasoning: support dual-currency (INR for stocks, USD for crypto)
-- without sparsely populated columns or JSONB hackery
CREATE TABLE portfolio_balances (
    id           UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    portfolio_id UUID NOT NULL REFERENCES portfolios(id) ON DELETE CASCADE,
    currency     TEXT NOT NULL CHECK (currency IN ('INR', 'USD')),
    cash_balance NUMERIC(18,2) NOT NULL DEFAULT 0 CHECK (cash_balance >= 0),
    UNIQUE (portfolio_id, currency)
);

-- 4. ORDERS
-- Reasoning: single orders table with polymorphic order types.
-- parent_order_id forms a linked-list for bracket orders (entry -> target, stop).
-- JSONB metadata captures type-specific fields without polluting the row.
-- Indexes:
--   (user_id, created_at DESC) -> trade history filtering
--   (status, asset_symbol)     -> matching engine polling for actionable orders
--   (parent_order_id)          -> bracket order cascade lookups
CREATE TYPE order_side AS ENUM ('buy', 'sell');
CREATE TYPE order_type AS ENUM ('market', 'limit', 'stop_loss', 'bracket');
CREATE TYPE order_status AS ENUM ('pending', 'open', 'filled', 'partially_filled',
                                  'cancelled', 'expired', 'rejected');

CREATE TABLE orders (
    id               UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id          UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    parent_order_id  UUID REFERENCES orders(id) ON DELETE SET NULL,
    side             order_side NOT NULL,
    order_type       order_type NOT NULL,
    asset_type       TEXT NOT NULL CHECK (asset_type IN ('stock', 'crypto')),
    asset_symbol     TEXT NOT NULL,  -- e.g. 'RELIANCE', 'BTCUSD'
    quantity         NUMERIC(18,8) NOT NULL CHECK (quantity > 0),
    price            NUMERIC(18,4),          -- limit price (limit/bracket)
    stop_price       NUMERIC(18,4),          -- trigger for stop-loss
    status           order_status NOT NULL DEFAULT 'pending',
    filled_quantity  NUMERIC(18,8) NOT NULL DEFAULT 0,
    avg_fill_price   NUMERIC(18,4),
    metadata         JSONB DEFAULT '{}',     -- bracket siblings, notes
    expires_at       TIMESTAMPTZ,
    created_at       TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at       TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
CREATE INDEX idx_orders_user_date    ON orders(user_id, created_at DESC);
CREATE INDEX idx_orders_pending      ON orders(status, asset_symbol)
    WHERE status IN ('pending', 'open');
CREATE INDEX idx_orders_parent       ON orders(parent_order_id)
    WHERE parent_order_id IS NOT NULL;

-- 5. POSITIONS
-- Reasoning: denormalized snapshot per (user, asset) for O(1) portfolio reads.
-- quantity > 0 = long; quantity < 0 would be short (optional, decide later).
-- Updated atomically on order fill via UPSERT.
CREATE TABLE positions (
    id               UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id          UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    asset_type       TEXT NOT NULL CHECK (asset_type IN ('stock', 'crypto')),
    asset_symbol     TEXT NOT NULL,
    quantity         NUMERIC(18,8) NOT NULL DEFAULT 0 CHECK (quantity >= 0),
    avg_buy_price    NUMERIC(18,4),
    -- current_price is denormalized, refreshed by price-update worker
    current_price    NUMERIC(18,4),
    created_at       TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at       TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    UNIQUE (user_id, asset_type, asset_symbol)
);

-- 6. WATCHLISTS
CREATE TABLE watchlists (
    id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id     UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    name        TEXT NOT NULL,
    created_at  TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    UNIQUE (user_id, name)
);

-- 7. WATCHLIST ITEMS
-- UNIQUE (watchlist_id, asset_symbol) prevents duplicate adds
CREATE TABLE watchlist_items (
    id           UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    watchlist_id UUID NOT NULL REFERENCES watchlists(id) ON DELETE CASCADE,
    asset_type   TEXT NOT NULL CHECK (asset_type IN ('stock', 'crypto')),
    asset_symbol TEXT NOT NULL,
    added_at     TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    UNIQUE (watchlist_id, asset_symbol)
);
CREATE INDEX idx_watchlist_items_wl ON watchlist_items(watchlist_id);

-- 8. PRICE ALERTS
-- Index on (user_id, asset_symbol, triggered) for fast "active alerts" scans
CREATE TABLE price_alerts (
    id           UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id      UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    asset_type   TEXT NOT NULL CHECK (asset_type IN ('stock', 'crypto')),
    asset_symbol TEXT NOT NULL,
    direction    TEXT NOT NULL CHECK (direction IN ('above', 'below')),
    target_price NUMERIC(18,4) NOT NULL,
    triggered    BOOLEAN NOT NULL DEFAULT FALSE,
    triggered_at TIMESTAMPTZ,
    created_at   TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
CREATE INDEX idx_price_alerts_active ON price_alerts(user_id, asset_symbol, triggered)
    WHERE triggered = FALSE;

-- 9. TRADE JOURNAL
-- Free-form notes + tags per closed trade
CREATE TABLE trade_journal (
    id           UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id      UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    order_id     UUID REFERENCES orders(id) ON DELETE SET NULL,
    asset_type   TEXT NOT NULL CHECK (asset_type IN ('stock', 'crypto')),
    asset_symbol TEXT NOT NULL,
    notes        TEXT,
    tags         TEXT[] DEFAULT '{}',
    exit_date    TIMESTAMPTZ,
    created_at   TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
CREATE INDEX idx_trade_journal_user ON trade_journal(user_id, created_at DESC);


Index reasoning summary
-----------------------

| Index                       | Why                                                                 |
|-----------------------------|----------------------------------------------------------------------|
| idx_portfolios_user         | UNIQUE -- enforces 1:1, fast lookup on login                         |
| portfolio_balances(portfolio_id, currency) | UNIQUE -- one balance per currency per portfolio       |
| idx_orders_user_date        | Covers 90% of user-facing queries (trade history, pagination)        |
| idx_orders_pending          | Partial index -- matching engine polls only pending/open orders, tiny |
| idx_orders_parent           | Partial -- bracket order cascade lookups (rare, so partial is fine)  |
| positions(user_id, asset_type, asset_symbol) | UNIQUE -- natural key for UPSERT on fill            |
| idx_price_alerts_active     | Partial -- only un-triggered alerts matter for the price-feed worker  |
| idx_watchlist_items_wl      | Covers "get all items in a watchlist"                                |
| idx_trade_journal_user      | User's journal sorted by recency                                     |


2. High-Level Architecture: Price Update Flow
----------------------------------------------

  +-----------------------------------------------------------------------+
  |                        EXTERNAL APIs                                  |
  |  [Alpha Vantage / Polygon.io / Yahoo Finance (mock for MVP)]          |
  +-----+---------------------------+--------------------------------------+
        | REST poll (every N       |
        | sec per batch of         |
        | symbols)                 |
        v                          v
  +---------------------+  +--------------------------------------------+
  |  Bull Queue          |  |  Order Matching Worker                     |
  |  (price-fetch queue) |  |  (reads idx_orders_pending,                |
  |                      |  |   checks conditions, writes                |
  |  Job per symbol      |  |   fills, updates positions)                |
  |  batch. Dedup by     |  |                                            |
  |  job ID = symbol.    |  +------------------+-------------------------+
  +----------+-----------+                     |
             | writes                          | writes fills
             v                                v
  +-------------------------------------------------------------------+
  |                         REDIS                                     |
  |  * Key-value: price:RELIANCE -> {price, change, ts, vol}          |
  |  * PUB/SUB channel: price:updates                                 |
  |  * Bull queue backend (job state, retry)                          |
  |  * Rate-limiter counters per user                                 |
  +----------------------+--------------------------------------------+
                         | PUBLISH
                         v
  +-------------------------------------------------------------------+
  |                    WEBSOCKET SERVER                                |
  |  * Subscribes to Redis PUB/Sub `price:updates`                    |
  |  * Maintains in-memory Map<symbol -> Set<wsConnection>>           |
  |  * On price event: lookup subscribers, send JSON frame            |
  |  * On client connect: send current snapshot from Redis             |
  +----------------------+--------------------------------------------+
                         | ws.send({symbol, price, change, sparkline})
                         v
  +-------------------------------------------------------------------+
  |              REACT CLIENT (Vite + Tailwind)                       |
  |  * WebSocket hook -> Zustand or useReducer store                   |
  |  * TradingView Lightweight Charts re-renders on price tick         |
  |  * Portfolio value recalculated reactively                        |
  +-------------------------------------------------------------------+


Key design choices in this flow:
- Price fetcher is a Bull queue job per symbol batch (not per symbol) to avoid
  overwhelming the external API rate limit. Single job runs every N seconds,
  fetches all watched symbols in one call if the API supports it (e.g.
  Polygon.io's snapshot endpoint).
- Redis acts as three things: price cache (fast reads for REST endpoints),
  pub/sub bus (decouples fetcher from WS server), and Bull backend.
- WebSocket server maintains per-symbol subscriber lists rather than broadcasting
  all prices to all clients -- essential for >100 concurrent users.


3. Three Riskiest Technical Decisions
--------------------------------------

Risk #1: PostgreSQL as the order-matching engine

The bet: Use Bull Queue workers that poll `orders` with
`WHERE status = 'pending'`, check price conditions against Redis, and execute
fills via SQL transactions.

Why it might break:
- Lock contention. A surge of limit orders at the same price level causes
  row-level locks on `positions` (hot row per symbol). At ~100 fills/sec,
  you'll see deadlocks or serialization failures.
- Eventual consistency gap. Bull Queue at-least-once delivery plus retries
  means a single limit order could trigger two fills if the worker crashes
  after `UPDATE orders` but before `INSERT INTO positions`. You need
  idempotency keys.
- No cross-order priority. If 1000 stop-loss orders trigger at the same price,
  PG has no built-in priority queue -- you'd need `SKIP LOCKED` +
  `ORDER BY created_at` which is O(n log n).

Tradeoff: Simpler to build (no new infra = Redis + PG + Bull is all you need)
but will need a dedicated, in-process matching engine (something like a L2
order book in Rust or Node with a lock-free ring buffer) at 500+ orders/day.
The schema is structured so you can swap the matcher later without data
migration.


Risk #2: WebSocket fan-out strategy tied to Node.js single thread

The bet: A single Node.js WebSocket server maintains an in-memory
`Map<symbol, Set<connection>>` and iterates to send messages.

Why it might break:
- CPU-bound broadcast. Sending one price update to 5000 subscribers means 5000
  `ws.send()` calls on the event loop -- blocking other connections during
  `JSON.stringify` and `write()`.
- Memory pressure. 10,000 connections each subscribing to 5 symbols = 50,000
  entries in the map + 10,000 socket objects. Node's heap will struggle around
  50k concurrent connections.
- Redis PUB/SUB fan-out. If you scale to 2 WS servers, Redis PUB/SUB delivers
  every price update to both -- doubling internal traffic.

Tradeoff: You can defer this to post-MVP because for a paper trading app,
concurrent active users will be low initially. When it becomes a problem,
migrate to:
  - uWebSockets.js (C++ addon, 10x throughput)
  - ioredis + per-connection Redis SUBSCRIBE (moves fan-out to Redis, but costs
    one Redis connection per WS client)
  - Or replace the whole stack with Centrifugo (Go-based, handles 100k
    connections natively)

The schema/subscription model (symbol-based rooms) is reusable regardless of WS
library choice.


Risk #3: Bracket order correctness (state machine complexity)

The bet: A bracket order is stored as a parent `order_type = 'bracket'` with
two child orders linked via `parent_order_id`. The matching worker handles
state transitions:
1. Entry leg fills -> place target (limit sell) + stop-loss (stop sell) as
   child orders
2. Target fills -> auto-cancel stop-loss
3. Stop-loss fills -> auto-cancel target

Why it might break:
- Race condition #1: Entry fills partially -> do you place child orders at the
  partial quantity or wait for full fill? Either choice leads to edge cases
  (what if the remaining partial fills later -- do you create additional child
  orders?).
- Race condition #2: Target and stop-loss conditions simultaneously satisfied
  in the same price tick. Deterministic resolution requires a total order on
  updates (e.g., process in order of order ID, or use a tick sequence number).
- Race condition #3: User manually cancels the parent bracket while the worker
  is processing a fill -- you need a compare-and-swap on `orders.status`
  (e.g., `UPDATE ... WHERE status = 'open'`).
- OCO (One-Cancels-Other) semantics require distributed locking or a saga
  pattern -- both are awkward in PostgreSQL + Bull Queue.

Tradeoff: Bracket orders are a marquee feature for a trading platform, so you
can't cut them. But the MVP should implement them sequentially (entry order
must be fully filled before child orders are placed) and warn users about
partial fills. Document that bracket behavior during high volatility may have a
1-2 second lag. The full state machine with partial-fill OCO is a v2
optimization.


Bottom Line for the MVP
-----------------------
The schema is solid and can survive all three refactors. The biggest immediate
risk is #3 (bracket correctness). Build market + limit + stop-loss first, get
those 100% right, then add brackets with the sequential-entry simplification.

Technologies used and WHY:

  Prisma (vs Knex):
    Native UUID PKs, ENUMs, JSONB, array types, partial indexes all map 1:1
    from the schema design. Declarative migrations (diff-based) catch subtle
    errors. Knex would require raw SQL for ENUMs and partial indexes.
    Tradeoff: Prisma struggles with complex CTEs/window functions, but this
    schema only needs CRUD + filters + aggregations.

  Express:
    Industry-standard Node.js HTTP framework. Minimal, well-understood, large
    middleware ecosystem (cors, rate-limit, helmet).

  Zod:
    Runtime type validation at the API boundary. Keeps validation out of route
    handlers and service layers. Provides clean error flattening for responses.

  ioredis:
    Mature Redis client with built-in retry strategy, lazy connection, and
    cluster support. Used here for OTP storage with TTL (SETEX command).

  bcryptjs:
    Pure-JS bcrypt implementation. No native compilation needed. 12 salt rounds
    balances security vs latency for a paper trading platform.

  jsonwebtoken:
    Standard JWT library. Signed with HS256 (symmetric). Access tokens carry
    {sub, email} and expire in 15m; refresh tokens expire in 7d.

How the auth flow works:

  1. User POSTs /auth/signup with {email, password, name}
  2. Backend hashes password with bcrypt (12 rounds)
  3. Prisma interactive transaction creates:
     a) User row (users table)
     b) Portfolio row (portfolios table, references user.id)
     c) Two portfolio_balance rows: INR = 1,000,000, USD = 0
  4. On success, returns {user, accessToken, refreshToken}
  5. On subsequent requests, client sends Authorization: Bearer <accessToken>
  6. middleware/auth.js verifies token signature + expiry
  7. Refresh tokens stored as httpOnly cookie, sent to /auth/refresh endpoint

  OTP 2FA flow:
  1. POST /auth/otp/generate -> generates 6-digit OTP, stores in Redis
     with SETEX (key: "otp:{email}", TTL: 300s, value: OTP)
  2. In production, OTP sent via email/SMS (MVP returns it in response)
  3. POST /auth/otp/verify -> compares OTP using crypto.timingSafeEqual
     (constant-time to prevent timing attacks). On success, deletes OTP
     from Redis (one-time use) and marks user.two_factor_enabled = true

  Why two tokens (access + refresh):
    - Short-lived access token (15m) limits theft damage window
    - httpOnly refresh token (7d) prevents XSS-based theft
    - Refresh tokens can be rotated (v2) for compromise detection
    - Single long-lived JWT cannot be revoked without a blocklist

  Why OTPs in Redis (not users table):
    - OTPs are ephemeral: auto-expire via TTL, no cleanup cron needed
    - O(1) lookups vs scanning a DB table
    - If Redis crashes, only in-flight OTP flows fail (no permanent data loss)

  Why transactional signup:
    - If user row is created but portfolio creation fails, orphan user exists
    - If portfolio is created but balance rows fail, portfolio has no balances
    - Prisma $transaction rolls back all 3 operations on any failure

  Why cooldown enforced at app layer (not DB):
    - CHECK constraints cannot reference NOW() (non-deterministic)
    - App checks cooldown_expires_at < NOW() in JavaScript
    - Race condition (two simultaneous resets) is idempotent for this case

  Why PrismaClient singleton in database.js:
    - PrismaClient manages a connection pool — expensive to re-create
    - Stored on globalThis to survive Node.js --watch hot reloads
    - Prevents connection leak on every file change during development

How the system works:

  CACHE-ASIDE PATTERN (for one-off requests):
    1. User requests price for RELIANCE
    2. Check Redis key "price:stock:RELIANCE"
    3. HIT -> return cached value (with stale flag if marked stale)
    4. MISS -> call Twelve Data API -> store result in Redis with 60s TTL -> return
    5. On rate limit: check "fallback" key (no TTL, written by poller) -> serve stale

  SELECTIVE BACKGROUND POLLER (for actively watched symbols):
    1. Every 10 seconds, Bull queue triggers pricePoller job
    2. Job queries DB for all distinct (assetType, assetSymbol) across
       watchlist_items, positions, and price_alerts tables
    3. Groups by type: { stock: Set["RELIANCE", "TCS"...], crypto: Set["BTCUSD"...] }
    4. For each type, calls the provider's batch endpoint ONCE for all symbols
       (Twelve Data supports comma-separated: ?symbol=RELIANCE,TCS,INFY)
    5. Writes each symbol's price to Redis cache (SETEX with 60s TTL)
    6. Also writes to a "fallback" key (SET with no TTL) for rate-limit resilience
    7. Publishes to Redis PUB/SUB channel "price:updates" for WebSocket fan-out

TTL Choice — Why 60 Seconds?

  - Too SHORT (< 15s): Every user refresh triggers an external API call.
    100 users viewing RELIANCE would hit Twelve Data's 800/day limit in ~8 minutes.
    The cache becomes useless — the 15s TTL just adds latency without reducing
    API calls.
  - Too LONG (> 300s): Portfolio values shown to users may be 5+ minutes stale.
    A user could see a portfolio value based on a price that's 5 minutes old.
    For a trading platform, even 60s is pushing it — but the poller (10s refresh)
    means active symbols are never actually limited by the 60s TTL.
  - 60 seconds is the sweet spot because:
    a) Active symbols: the poller overwrites the cache every 10s, so the TTL
       barely matters — they always have fresh data.
    b) Inactive symbols: a one-off lookup stays cached for 60s. If another user
       looks up the same symbol within 60s, they get a cache hit. After 60s with
       no activity, the key auto-expires (self-cleaning).
    c) Cost: ~1 API call per symbol per minute maximum. With 200 active symbols,
       that's 200 calls/min minimum, but batched into ~2 API calls (1 for stocks,
       1 for crypto) = only 2 calls/min!

Why Refreshing Everything Would Be Wasteful

  - Twelve Data free tier: ~800 requests/day
  - NSE alone has ~2,000 liquid stocks. CoinGecko tracks ~12,000 coins.
  - Polling all 14,000 symbols every 60 seconds: 14,000 * 60 * 24 = 20M calls/day
  - Even polling just NSE 500 every 10 seconds: 500 * 6 * 60 * 24 = 4.3M calls/day
  - Both are orders of magnitude beyond any free/freemium API tier.
  - Our approach: query the DB for distinct symbols across watchlist_items,
    positions, and price_alerts. For a platform with 100 users each watching
    10 symbols + holding 5 positions, that's ~1,500 symbols. At 1 batch API call
    per provider per 10s = 6 calls/min. ~8,640 calls/day — within Twelve Data's
    Pro tier but over the free 800/day. So for MVP with Twelve Data free, the
    poller interval should be reduced to ~60s for stocks (800/24/60 = ~0.55/min).
    The system automatically handles this via the rate limit tracking: if the
    limit is hit, it marks data as stale and retries on the next cycle.

Rate Limit Handling

  When the external API rate limit is hit:
  1. The provider throws RateLimitError (detected via status 429 or Redis counter)
  2. The cache-aside path: serves data from the "fallback" Redis key (set by the
     background poller with no TTL) and sets `stale: true` and `rateLimited: true`
  3. The background poller: skips the current cycle, logs a warning, and tries
     again on the next tick (10s later)
  4. The HTTP response sets headers `X-Data-Stale: true` and `X-RateLimit-Provider`
     so the frontend can show a warning banner
  5. The Redis rate limit counter auto-expires after 60 seconds (TTL on the key),
     so the next poll cycle will naturally resume calling the API

  Why this is safe:
  - No crash: the app never throws an unhandled error from rate limiting
  - No empty data: the fallback key ensures there's always SOME price to show
  - Self-healing: the Redis TTL on the rate limit counter auto-resets
  - Visible staleness: the frontend can show "Stale data — API rate limited"
    so the user knows the price isn't current

Core Scalability Lesson: Cache-Aside + Selective Refresh

  This design avoids rate limits even with many concurrent users because:

  1. AMORTIZATION: One API call serves N users for M seconds.
     - User A requests RELIANCE: cache miss -> API call -> cache populated
     - Users B, C, D request RELIANCE in the next 60s: all cache hits -> 0 API calls
     - With 100 users all viewing RELIANCE: 1 API call per minute, not 100

  2. BATCHING: One API call serves all symbols for one provider.
     - Twelve Data batch: ?symbol=RELIANCE,TCS,INFY,... (up to ~100 per call)
     - Instead of 100 separate API calls, we make 1
     - CoinGecko batch: ?ids=bitcoin,ethereum,... (up to ~250 per call)
     - Instead of 50 separate API calls, we make 1

  3. SELECTIVITY: Only poll symbols users actually care about.
     - Market has 14,000+ symbols. Users watch ~1,500.
     - Polling 1,500 vs 14,000 = 9.3x fewer API calls
     - Using batch endpoint: 2 calls vs ~1,500 calls = 750x fewer

  4. CASCADE OF CACHES:
     - L1: Redis price cache (60s TTL) — absorbs user traffic
     - L2: Fallback key (no TTL) — absorbs rate limit events
     - L3: Stale marker — signals to the frontend that data may be old
     - No layer ever crashes — each degrades gracefully to the next

  Combined effect: 100 users refreshing every 5 seconds for 10 symbols each =
  1,000 req/s to the backend. But the backend makes only 2 API calls every
  10 seconds (one batch for stocks, one batch for crypto). That's ~0.2 API
  calls per second serving 1,000 user requests per second. A 5,000x
  amplification factor.

WHY PER-SYMBOL ROOMS (NOT ONE GLOBAL CHANNEL)?

  With a global broadcast channel, every client receives EVERY price update for
  EVERY symbol — 14,000+ messages per second. The client filters to find the 5
  symbols it cares about. This is wasteful at every layer:

    Client CPU:    JS thread processes 14,000 irrelevant messages/sec
    Network:       10,000 clients x 14,000 updates = 140M messages/sec
    Server memory: Socket.io tracks delivery state per-client for backpressure

  With per-symbol rooms (Socket.io built-in):
    - Client A watching RELIANCE → server adds to room "price:stock:RELIANCE"
    - Client B watching BTCUSD → server adds to room "price:crypto:BTCUSD"
    - RELIANCE price tick → server emits to room "price:stock:RELIANCE"
    - Client A: 1 message. Client B: 0 messages.
    - Each client receives EXACTLY N messages per tick, where N = symbols watched.
    - O(N) per client instead of O(14,000). Linear vs linear — the difference
      between 5 and 14,000 is a 2,800x reduction.

SUBSCRIBE/UNSUBSCRIBE LIFECYCLE:

  1. User lands on /watchlist → component renders with subscription list
  2. useSocketPrice effect runs → socket.emit("subscribe", 10 symbols)
  3. Server adds client to 10 rooms → starts receiving relevant updates
  4. User clicks on RELIANCE → navigates to /asset/RELIANCE
  5. Old component unmounts: cleanup unsubscribes the 10 watchlist symbols
  6. New component mounts: subscribes to just ["RELIANCE"]
  7. After 5 seconds, user goes back to /watchlist → the reverse happens
  8. Result: at any moment, the server only pushes updates the user cares about

  If we DIDN'T unsubscribe on navigation: user views 50 assets over a session.
  The server accumulates 50 subscriptions. Every price tick pushes 50 updates
  to that client. With 10,000 users doing this: 500,000 updates/tick. The server
  collapses under the weight. Subscriptions must be EXACTLY the current viewport.

RECONNECTION STRATEGY:

  Socket.io built-in reconnection with exponential backoff:

    Attempt 1: 1000ms
    Attempt 2: 2000ms
    Attempt 3: 4000ms
    Attempt 4: 8000ms
    Attempt 5: 16000ms
    Attempt 6+: 30000ms (cap)
    Jitter: delay randomized ±50% (randomizationFactor: 0.5)

  Why exponential backoff (not fixed):
    - Fixed 1s retry: if server is down 60s, 60 useless attempts per client.
      10,000 clients = 600,000 attempts flooding server logs on restart.
    - Exponential: fast retry for brief glitches, then backs off.
    - Jitter: prevents thundering herd when 10,000 clients all retry at the
      same time after a server restart.

  What happens on reconnect:
    1. Socket.io fires "connect" event
    2. useSocketPrice's handleReconnect fires
    3. Reads current subscriptions from the ref (always up-to-date)
    4. Emits "subscribe" for all of them
    5. Server adds client to rooms → price updates resume
    6. User never misses updates — the delay is just the reconnect window

  Why useRef for the reconnect handler (instead of effect closure):
    - The effect captures subscriptions at mount time
    - If the effect doesn't re-run (subscriptions unchanged), the captured
      value is stale
    - subsRef.current always has the latest value from the render cycle
    - On reconnect, we always subscribe to what the user CURRENTLY sees

TECHNOLOGY CHOICES:

  Socket.io vs raw WebSocket:
    - Socket.io provides rooms (server-side filtering) built-in
    - Raw WebSocket requires manually tracking room membership
    - Socket.io provides reconnection, heartbeats (pingInterval/pingTimeout),
      and event-based messaging out of the box
    - The overhead (https://github.com/socketio/socket.io-benchmarks) is ~2%
      vs raw WebSocket — negligible for a trading app

  Two Redis connections (redis + redisSubscriber):
    - Redis protocol limitation: SUBSCRIBE mode blocks all other commands
    - ioredis enforces this: same connection can't mix SUB/UNSUB with GET/SET
    - redis: regular commands + Bull queue
    - redisSubscriber: SUBSCRIBE "price:updates" only
    - Cost: one extra TCP connection (~0.1ms per message). Benefit: clean
      separation of concerns.

  One socket per app (not per component):
    - Each WebSocket connection costs ~10-50KB server memory
    - 10 components x 10 connections = 100 connections = 1-5MB
    - 10 components x 1 connection (rooms) = 1 connection = 50KB
    - Socket.io multiplexes events over one TCP connection — rooms are
      server-side data structures, not separate TCP sockets

SYNC vs ASYNC — CORE DESIGN DISTINCTION:

  MARKET ORDERS (SYNCHRONOUS):
    The user gets an immediate response. The order is executed in the request
    handler — no queue, no delay.

    Flow:
      Client POST /orders/market → server gets cached price → DB transaction:
        cash check → cash deduct → position UPSERT → order row → HTTP 201 response

    Latency: ~50ms (Redis read) + ~20ms (DB transaction) = ~70ms total.
    The user sees "FILLED" immediately.

    Why this works: market orders don't need price monitoring. They take whatever
    the current price is. The only question is "do you have enough cash/position?"
    — which is a local check against the user's portfolio, not a market condition.

  LIMIT / STOP-LOSS / BRACKET ORDERS (EVENT-DRIVEN ASYNC):
    The user gets an HTTP 201 with status="pending". The order is placed in the
    database and picked up later by the order matcher worker.

    Flow:
      Client POST /orders/limit → INSERT order status="pending" → HTTP 201(pending)
      ...later, price tick for this symbol arrives...
      priceBroadcast.js → queueOrderCheck() → Bull job {symbol, assetType}
      orderMatcher worker → query pending orders for symbol → shouldExecute()?
        Yes → executeFill() in transaction → order status="filled"
      (Order fill result is pushed to client via WebSocket in a future phase)

    The key insight: async orders WAIT for an external condition (price reaching
    a level). The server can't predict when that will happen. Blocking the HTTP
    request until the condition is met would:
      - Hold open 10,000+ HTTP connections for hours
      - Tie up Node.js event loop threads
      - Time out browsers (most HTTP clients timeout after 30-120 seconds)
      - Crash the server on memory (each pending HTTP request ~50KB)

  WHY THIS DISTINCTION MATTERS:
    Synchronous execution is for operations with KNOWN, CURRENT inputs.
    Asynchronous execution is for operations that depend on FUTURE, UNKNOWN inputs.

    In a trading platform:
      Market order = known current price → sync (70ms)
      Limit order = unknown future price → async (minutes/hours/days)

    Getting this wrong (making market orders async) adds unnecessary 70ms latency
    to every trade. Getting it wrong the other way (making limit orders sync)
    would require holding HTTP connections open for potentially days.

RACE CONDITION HANDLING — PESSIMISTIC ROW LOCKING:

  We chose PESSIMISTIC locking (SELECT FOR UPDATE / conditional UPDATE) over
  optimistic locking (version + retry). Here's why:

  THE PROBLEM:
    User has ₹100,000 cash. Two market buy orders arrive simultaneously:
      Order A: buy RELIANCE ₹60,000
      Order B: buy TCS ₹60,000

    Without protection, both orders read cash_balance = ₹100,000, both see
    "sufficient funds", both deduct ₹60,000 → cash_balance = -₹20,000.

  OPTIMISTIC LOCKING (REJECTED):
    - Add a version column to portfolio_balances
    - Read: version=5, cash=₹100,000
    - Write: UPDATE ... SET cash=₹40,000, version=6 WHERE id=X AND version=5
    - If version changed (another order committed first), retry the whole tx
    - PROBLEM: On retry, Order B reads cash=₹40,000 which is < ₹60,000. It
      fails anyway. The retry was wasted work. For portfolio operations where
      contention is high, optimistic locking causes many wasteful retries.

  PESSIMISTIC LOCKING (CHOSEN):
    - For BUYS: `UPDATE portfolio_balances SET cash = cash - amount WHERE id = X AND cash >= amount`
      PostgreSQL's UPDATE acquires a ROW EXCLUSIVE lock on the row.
      Transaction B's UPDATE queues behind A's lock.
      When A commits, B reads the NEW cash balance (₹40,000).
      ₹40,000 < ₹60,000 → 0 rows updated → "Insufficient cash balance".
      No retry needed. No wasted work.
    - For SELLS: `SELECT quantity FROM positions WHERE ... FOR UPDATE`
      Explicit row lock on the position row prevents two concurrent sells
      from reading the same quantity and overselling.

  WHY PESSIMISTIC IS BETTER HERE:
    1. High contention: the cash balance row is a single point of contention
       for all orders by the same user. Optimistic locking would retry often.
    2. Predictable latency: pessimistic locking makes each request wait its
       turn (max wait = time for one transaction ~20ms). No exponential retry.
    3. No version column: the conditional UPDATE itself is the check — no
       schema changes needed.
    4. PostgreSQL's row locks are cheap: ~100 bytes per locked row, deadlock
       detection handles rare circular waits.


THE FULL ORDER LIFECYCLE — System Design Interview Explanation:

  ┌─────────────────────────────────────────────────────────────────────────┐
  │              ORDER LIFECYCLE — FROM SUBMISSION TO FILL                  │
  └─────────────────────────────────────────────────────────────────────────┘

  PHASE 1: ORDER SUBMISSION

    Client POSTs to one of:
      /orders/market   → SYNC path
      /orders/limit    → ASYNC path
      /orders/stop-loss
      /orders/bracket

    MARKET PATH (SYNC):
      1. Read current price from Redis cache (O(1), ~1ms)
         Key: "price:stock:RELIANCE" → { price: 2500, ... }
      2. Calculate total = quantity × price
      3. Enter Prisma $transaction (one DB round-trip):
         a. Atomic cash check-and-deduct:
            UPDATE portfolio_balances
            SET cash_balance = cash_balance - total
            WHERE id = X AND cash_balance >= total
            PostgreSQL row lock serializes concurrent deductions.
            If 0 rows updated → INSUFFICIENT CASH → throw → ROLLBACK.
         b. UPSERT position:
            INSERT INTO positions (user, asset, qty, avg_price)
            VALUES (...) ON CONFLICT DO UPDATE SET
            quantity = old.qty + new.qty,
            avg_buy_price = weighted average
         c. INSERT INTO orders (status='filled', filled_quantity=qty, avg_fill_price=price)
      4. Return order to client with status='filled' → HTTP 201 (70ms total)
      5. Client shows: "Bought 10 RELIANCE @ ₹2,500 — FILLED"

    LIMIT / STOP-LOSS PATH (ASYNC):
      1. Validate inputs (price > 0, quantity > 0)
      2. INSERT INTO orders (status='pending', order_type='limit', ...)
      3. Return order to client with status='pending' → HTTP 201
      4. Client shows: "Limit order placed — PENDING"

    BRACKET PATH (ASYNC, TRANSACTIONAL):
      1. Validate: target on correct side, stop on correct side
      2. Prisma $transaction:
         a. INSERT parent order (order_type='bracket', status='pending')
         b. INSERT target child (order_type='limit', parent_order_id=parent.id, status='pending')
         c. INSERT stop-child (order_type='stop_loss', parent_order_id=parent.id, status='pending')
      3. Return parent order → HTTP 201
      4. Client shows: "Bracket order placed — PENDING"

  PHASE 2: ORDER MONITORING (ASYNC ORDERS ONLY)

    Every price update triggers this chain:
      1. Price poller → updatePriceCache() → Redis PUB/SUB "price:updates"

      2. priceBroadcast.js receives the message:
         a. Socket.io: forward to clients in room "price:stock:RELIANCE"
         b. Bull queue: queueOrderCheck("stock", "RELIANCE")

      3. orderMatcher worker picks up the Bull job:
         a. Query: SELECT * FROM orders
                   WHERE asset_type='stock'
                     AND asset_symbol='RELIANCE'
                     AND status IN ('pending', 'open')
                     AND order_type IN ('limit', 'stop_loss')
                     AND parent_order_id IS NULL
                   ORDER BY created_at ASC
         b. For each order, read current price from Redis → shouldExecute()?

      CONDITION CHECK (shouldExecute):
        Limit buy:  market_price <= limit_price → EXECUTE
        Limit sell: market_price >= limit_price → EXECUTE
        Stop-loss buy:  market_price >= stop_price → EXECUTE
        Stop-loss sell: market_price <= stop_price → EXECUTE
        Bracket entry: same as limit order (entry leg is a limit order)

      FALLBACK: A full scan runs every 60 seconds, checking ALL pending orders
      regardless of whether their symbol had a recent price update. This catches
      any orders missed by the event-driven path (e.g., race between order creation
      and PUB/SUB delivery).

  PHASE 3: ORDER EXECUTION (FILL)

    When shouldExecute() returns true, the worker calls executeFill():

      executeFill(userId, { side, assetType, symbol, qty, price }, orderId):
        → Same code path as market orders

        → Prisma $transaction:
          1. JOIN portfolio + portfolio_balance, SELECT FOR UPDATE on balance
             (prevents concurrent fills from double-deducting cash)
          2. BUY: UPDATE cash SET cash = cash - total WHERE cash >= total
                  If fail → "Insufficient cash" → rollback → order stays pending
          3. BUY: UPSERT position (qty += new, avg = weighted avg)
          4. SELL: SELECT FOR UPDATE on position
                   If qty < sell_qty → "Insufficient position" → rollback
          5. SELL: UPDATE position SET qty = qty - sell
          6. SELL: UPDATE cash SET cash = cash + total
          7. UPDATE orders SET status='filled', filled_quantity=qty, avg_fill_price=price
          8. RETURN order

      BRACKET CHILD HANDLING (after entry leg fills):
        1. armBracketChildren(parentOrderId):
           UPDATE orders SET status='open' WHERE parent_order_id = X AND status='pending'
        2. Both children now visible to the order matcher
        3. On subsequent price ticks, worker checks children → fills target or stop

      OCO (One-Cancels-Other) — after a bracket child fills:
        1. cancelSiblingOrder(filledChildId):
           Find sibling: same parent, different child, status IN (pending, open)
           UPDATE orders SET status='cancelled' WHERE id = sibling_id
        2. Prevents both target AND stop-loss from filling

  PHASE 4: POST-FILL

    [Future phase: WebSocket push of fill event to the user]
    The filled order appears in GET /orders?status=filled
    Portfolio value updates reactively (positions changed → recalculate)


WHY THE DESIGNS DIFFER — SYSTEM DESIGN LESSON:

  Market orders = synchronous because:
    - Input is known NOW (cached price)
    - No external dependency (no waiting for price to move)
    - User expects immediate feedback ("did my trade go through?")
    - Can complete in one HTTP request-response cycle (~70ms)
    - Failure mode: user sees error immediately, can retry

  Limit/stop orders = async/event-driven because:
    - Input is UNKNOWN (future price)
    - External dependency (market movement)
    - Can't hold HTTP connection open for hours
    - Bull queue provides durability (job survives server restart)
    - Failure mode: order stays pending, retries on next price tick
    - Event-driven is more efficient than polling (orders checked only when
      their symbol's price moves, not every N seconds)

  The dividing line: if the operation depends on information the server
  ALREADY HAS (cached price, portfolio state), execute sync. If it depends
  on information the server DOESN'T HAVE YET (future price), execute async.

WHY PRICE ALERTS REUSE THE SAME WORKER PATTERN AS ORDERS:

  Both pending orders and price alerts are structurally identical:

    ┌──────────────┬────────────────────┬───────────────────────────────┐
    │              │ Pending Order      │ Price Alert                   │
    ├──────────────┼────────────────────┼───────────────────────────────┤
    │ DB row       │ status='pending'   │ triggered=false               │
    │ Condition    │ market_price       │ market_price crosses          │
    │              │ crosses limit      │ target_price                  │
    │ Evaluation   │ On every price     │ On every price                │
    │              │ tick for symbol    │ tick for symbol               │
    │ Action       │ UPDATE status=     │ UPDATE triggered=true,        │
    │              │ 'filled' + cash/   │ store notification            │
    │              │ position mutation  │                               │
    │ Failure mode │ Skipped, retries   │ Skipped, retries              │
    │              │ next tick          │ next tick                     │
    └──────────────┴────────────────────┴───────────────────────────────┘

  If we built a separate system for alerts (e.g., a cron job scanning the alert
  table every 30 seconds), we'd have DUPLICATE INFRASTRUCTURE:

    - Two polling loops instead of one event-driven trigger
    - Two sets of Redis connections, Bull queues, error handlers
    - Two failure modes to monitor
    - The cron would scan thousands of rows every cycle even when no prices
      have changed — wasting CPU and DB IO

  By reusing the same event-driven Bull queue pattern:
    1. One price update → one Redis PUB/SUB message
    2. priceBroadcast.js pushes TWO Bull jobs (one for orders, one for alerts)
    3. Both workers run the same loop: query rows → check condition → execute
    4. Both share the 60s fallback scan for safety
    5. If Redis PUB/SUB fails, both degrade together (gracefully, via fallback)
    6. Adding a new condition-checking feature (e.g., "technical indicator
       trigger") would follow the exact same pattern — just another Bull queue
       with the same structure


TRADE HISTORY INDEXING — COLUMN CHOICES:

  The trade history query pattern is:
    SELECT * FROM orders
    WHERE user_id = ?
      [AND status = ?]
      [AND asset_symbol = ?]
      [AND order_type = ?]
      [AND created_at BETWEEN ? AND ?]
    ORDER BY created_at DESC
    LIMIT 20 OFFSET ?

  The existing index `idx_orders_user_date (user_id, created_at DESC)` covers the
  BASE query (no filters, just chronological). But adding filters breaks index
  usefulness:

    WITHOUT a (user_id, status) index:
      PostgreSQL scans ALL rows matching user_id, then filters by status.
      If the user has 1,000 orders and 10 are "filled", we scan 1,000 rows
      to find 10. That's 99% wasted work.

    WITH a (user_id, status, created_at DESC) index:
      PostgreSQL seeks directly to the intersection of user_id AND status.
      For "show me filled orders": seeks the 10 "filled" rows directly.
      Zero wasted work.

  The same logic applies to asset_symbol (common filter: "show me my RELIANCE
  trades") and order_type ("show me my limit orders").

  The index order matters: (user_id, status, created_at) is correct because:
    - user_id has highest selectivity (reduces to one user)
    - status has medium selectivity (filters to ~3-5 values)
    - created_at provides the ORDER BY without an extra sort step

  We did NOT index (asset_symbol, status) standalone because:
    - These queries are always scoped to one user (WHERE user_id = ?)
    - An index without user_id as the leading column would scan ALL users'
      orders for the asset — useless for multi-tenant data
    - PostgreSQL can still use the (user_id, asset_symbol) index efficiently

ON-THE-FLY vs PRECOMPUTE — DECISION:

  We chose ON-THE-FLY with a 30-second Redis cache.

  WHY NOT PRECOMPUTE (write to DB after every trade):
    Portfolio value changes on EVERY PRICE TICK (every 10 seconds for active symbols).
    If we precomputed P&L after every trade, we'd also need to recompute on every
    price tick. That's 6 recomputations per minute per user × 100 users = 600 writes
    per minute. And 99% of those writes are never read (no one is looking at the
    dashboard that instant). That's wasted CPU and DB IO.

  WHY NOT FULLY ON-THE-FLY (no cache):
    P&L history queries scan the orders table (up to thousands of rows per user).
    If a user refreshes the dashboard 5 times in 10 seconds, each refresh does the
    same scan. With 100 users each doing this, that's 500 scans/min × 1000 rows = 500K
    rows scanned per minute for no benefit.

  WHY 30-SECOND CACHE IS THE SWEET SPOT:
    - Trades are infrequent (5-10 per user per day). After a fill, max staleness = 30s.
    - Price changes: positions' current_price updates every 10s, but analytics lags
      by at most 30s. For a paper trading platform, this is acceptable.
    - Cache absorbs rapid refreshes: CTRL+R spam doesn't hit the DB.
    - Self-cleaning: 30s TTL means no stale data accumulates.
    - No explicit invalidation needed: the TTL handles everything.

  INVALIDATION (what we WOULD do if precomputing):
    If we precomputed, we'd need to invalidate on:
    1. Every order fill (market, limit triggered, etc.)
    2. Every price tick (since positions value changed)
    This is more invalidation triggers than actual reads. The TTL-based approach
    avoids this complexity entirely.

  TRADEOFF ACCEPTED:
    After a trade fill, the dashboard may show stale data for up to 30 seconds.
    For a paper trading platform with no real money at risk, this is acceptable.
    The user can always refresh the page to see current data (if the 30s cache
    hasn't expired, they'd still see stale data — but refreshing again after 30s
    would show fresh data).


METRIC COMPUTATION DETAILS:

  P&L (Total):
    current_value = cash + Σ(position_qty × current_price)
    total_pnl = current_value − initial_deposit

  P&L (Daily):
    For each day: realized P&L = Σ(sell_credit) − Σ(buy_cost) for fills that day.
    Cumulative P&L = sum of daily realized P&L up to that day.
    NOTE: This is REALIZED P&L only. Unrealized gains from open positions are
    NOT included in daily breakdowns. The total P&L (from portfolio summary)
    includes unrealized gains.

  Win Rate:
    profittable_trades / total_closed_trades
    For MVP: compares sell price to the average buy price (not FIFO matched).
    This overestimates the win rate slightly but is a reasonable approximation.

  Best/Worst Asset:
    Positions ranked by (quantity × (current_price − avg_buy_price)).
    Includes only open positions with quantity > 0.

  Sector Allocation:
    For MVP: grouped by asset_type only ("stock" vs "crypto").
    A production version would use a symbol→sector reference table
    (e.g., RELIANCE → "Energy", TCS → "Technology").

  Average Holding Period:
    For each closed position: days between first buy and last sell of that symbol.
    Approximate — does not reflect partial lot matching (FIFO/LIFO).

  Volatility:
    Annualized = std_dev(daily_returns) × √252
    daily_return = day_pnl / running_portfolio_value

  Sharpe Ratio:
    (annualized_portfolio_return − risk_free_rate) / annualized_volatility
    risk_free_rate = 6% (approximate Indian risk-free rate)
    Requires at least 2 days of trade history.

DECISIONS:

  Skeleton-based Layout:
    Most trading platforms use a sidebar. We made it collapsible (icon-only at 56px)
    for screen space efficiency. Firefox/full-screen users can collapse to see more charts.
    Uses NavLink for active-state highlighting (emerald glow).

  Auth token in localStorage (not httpOnly cookie):
    Tradeoff: httpOnly cookie is safer (XSS-proof), but requires same-origin or BFF.
    localStorage is simpler for MVP and avoids a BFF proxy layer. Risk accepted — MVP
    has no sensitive PII. Migration path: add backend-for-frontend with cookie-based
    refresh token, keep localStorage access token.

  Fetch wrapper over axios:
    We wrote api.js using fetch (native, no extra dep) instead of installing axios.
    The key feature: 401 auto-refresh with token rotation. This avoids the user
    getting logged out mid-session when the access token expires.

  TradingView Lightweight Charts (not TradingView Advanced Charts):
    Lightweight Charts is free, open-source, and has zero external dependencies.
    Advanced Charts requires a TradingView license ($). Charts API is minimal:
    createChart() + addLineSeries() + setData(). For MVP candlesticks and volume
    are unnecessary — line charts show price action adequately.

  Recharts for dashboard charts (not Chart.js or D3):
    Recharts is React-native (JSX-based), integrates directly with React state,
    and has a ResponsiveContainer that auto-resizes. Chart.js requires canvas refs
    and imperative API calls. D3 is too low-level for 3 simple charts.

  Analytics dashboard fetches all 8 metric routes in one call:
    The GET /analytics/dashboard endpoint returns summary, PnL history, win rate,
    best/worst, sectors, holding period, volatility, and Sharpe in a single response.
    This avoids 8 sequential requests and simplifies the page component.

  AssetDetail page has its own history fetch:
    /market/history/:type/:symbol is a new conceptual endpoint (not yet implemented
    in backend). The page will gracefully show "No historical data available" if
    the endpoint 404s. The WebSocket price updates will still work, keeping the
    current price accurate even without historical data.
